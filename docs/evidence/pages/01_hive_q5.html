SELECT month,COUNT(*) contacts,100.0*SUM(CASE WHEN y='yes' THEN 1 ELSE 0 END)/COUNT(*) subscription_pct
FROM client_info GROUP BY month ORDER BY contacts DESC LIMIT 1;
